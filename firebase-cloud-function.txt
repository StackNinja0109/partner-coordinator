const _ = require('lodash');
const { IncomingWebhook } = require('@slack/webhook');

require('dotenv').config()

const { onDocumentWritten } = require("firebase-functions/v2/firestore");
const { setGlobalOptions } = require('firebase-functions/v2');

const { Client } = require('@elastic/elasticsearch')

setGlobalOptions({ region: 'asia-northeast1' })


const eCloudId = process.env.ELASTIC_CLOUDID
const eCloudPass = process.env.ELASTIC_PASS
const projectId = process.env.GCLOUD_PROJECT

const elasticSearchClient = new Client({
    cloud: {
        id: eCloudId
    },
    auth: {
        username: 'elastic',
        password: eCloudPass
    }
})

const slackNotification = (message) => {
    const url = process.env.SLACK_WEBHOOK_URL;
    const webhook = new IncomingWebhook(url);
    webhook.send({
        text: `Cloud Function Notification: ${message}`,
      });
}

exports.customerElasticIndexor = onDocumentWritten("cust/{custId}", (event) => {
    const elasticIndex = `${projectId}-customer`
    const newValue = event.data.after.data();
    const beforeValue = event.data.before.data()
    const documentId = event.params.custId;

    if (newValue === undefined) {
        // removed document
        console.log(`${elasticIndex}: document deleted ${documentId}`)
        elasticSearchClient.delete({ index: elasticIndex, id: documentId })
    } else if (beforeValue === undefined) {
        // document created newly
        console.log(`${elasticIndex}: document created ${documentId}`)
        if (Object.keys(newValue).length !== 0) {
            const res = elasticSearchClient.index({ index: elasticIndex, id: documentId, body: newValue })
            res.catch((err) => {
                slackNotification(`${elasticIndex}:${documentId}: ${err}`)
            })
        }
    } else {
        //  document updated
        if (!_.isEqual(beforeValue, newValue)) {
            if (Object.keys(newValue).length !== 0) {
                console.log(`${elasticIndex}: document updated ${documentId}`)
                const res = elasticSearchClient.index({ index: elasticIndex, id: documentId, body: newValue })
                res.catch((err) => {
                    slackNotification(`${elasticIndex}:${documentId}: ${err}`)
                })
            }
        }
    }
});
